# Примеры использования нормализации названий организаций

## Быстрый старт

```python
from src.postprocessing.data_normalizer import DataNormalizer

# Создание экземпляра нормализатора
normalizer = DataNormalizer()

# Нормализация названия организации
result = normalizer._normalize_organization_name("ООО Экомед")
print(result)  # Выведет: "Экомед"
```

## Примеры нормализации

### 1. Удаление юридических форм

```python
# Российские формы
normalizer._normalize_organization_name("ООО Экомед")           # → "Экомед"
normalizer._normalize_organization_name("ПАО Газпром")          # → "Газпром"
normalizer._normalize_organization_name("ЗАО Рога и копыта")    # → "Рога и копыта"
normalizer._normalize_organization_name("АО Сбербанк")          # → "Сбербанк"

# Медицинские учреждения
normalizer._normalize_organization_name("ФБУЗ Центр гигиены")   # → "Центр гигиены"
normalizer._normalize_organization_name("ГБУЗ Поликлиника №1")  # → "Поликлиника №1"

# Международные формы
normalizer._normalize_organization_name("Microsoft LLC")        # → "Microsoft"
normalizer._normalize_organization_name("Apple Inc")           # → "Apple"
```

### 2. Сохранение ИП

```python
# ИП сохраняют префикс
normalizer._normalize_organization_name("ИП Иванов")                    # → "ИП Иванов"
normalizer._normalize_organization_name("ИП Петров Сергей")             # → "ИП Петров Сергей"
normalizer._normalize_organization_name("ип сидоров")                   # → "ИП сидоров"
normalizer._normalize_organization_name("ИП Козлова Анна Петровна")     # → "ИП Козлова Анна Петровна"
```

### 3. Обработка кавычек

```python
# Кавычки-ёлочки
normalizer._normalize_organization_name("ООО «Экомед»")         # → "Экомед"
normalizer._normalize_organization_name("ПАО «Газпром»")        # → "Газпром"

# Прямые кавычки
normalizer._normalize_organization_name('ООО "Рога и копыта"')  # → "Рога и копыта"
normalizer._normalize_organization_name('ИП "Иванов"')          # → "ИП Иванов"

# Одинарные кавычки
normalizer._normalize_organization_name("АО 'Сбербанк'")        # → "Сбербанк"
```

### 4. Сложные случаи

```python
# Множественные пробелы
normalizer._normalize_organization_name("ООО    Экомед   ")     # → "Экомед"

# Смешанный регистр
normalizer._normalize_organization_name("ооо ЭКОМЕД")           # → "Экомед"
normalizer._normalize_organization_name("пао газпром")          # → "Газпром"

# Без юридических форм
normalizer._normalize_organization_name("Экомед")               # → "Экомед"
normalizer._normalize_organization_name("центр медицины")       # → "Центр медицины"

# Пустые значения
normalizer._normalize_organization_name("")                     # → ""
normalizer._normalize_organization_name(None)                   # → ""
normalizer._normalize_organization_name("   ")                  # → ""
```

## Интеграция в обработку данных

```python
# Обработка списка организаций

    1: {"name": "ООО Экомед", "city": "Красноярск"},
    2: {"name": "ИП Иванов Петр", "city": "Москва"},

# Нормализация всех организаций

# 3: Газпром
```

## Обработка ошибок

```python
def safe_normalize_organization_name(normalizer, name):
    """Безопасная нормализация с обработкой ошибок"""
    try:
        if not name or not isinstance(name, str):
            return ""
        
        result = normalizer._normalize_organization_name(name)
        return result if result else name  # Возвращаем оригинал, если результат пустой
        
    except Exception as e:
        print(f"Ошибка при нормализации '{name}': {e}")
        return name  # Возвращаем оригинальное название при ошибке

# Использование
normalizer = DataNormalizer()
safe_result = safe_normalize_organization_name(normalizer, "ООО Экомед")
```

## Настройка для специфических случаев

```python
class CustomDataNormalizer(DataNormalizer):
    """Расширенный нормализатор с дополнительными правилами"""
    
    def _normalize_organization_name(self, name: str) -> str:
        # Сначала применяем базовую нормализацию
        result = super()._normalize_organization_name(name)
        
        # Добавляем специфические правила
        custom_replacements = {
            "Рога и копыта": "РиК",
            "Центр гигиены": "ЦГ"
        }
        
        for old, new in custom_replacements.items():
            if result == old:
                result = new
                break
        
        return result

# Использование кастомного нормализатора
custom_normalizer = CustomDataNormalizer()
result = custom_normalizer._normalize_organization_name("ФБУЗ Центр гигиены")
print(result)  # Выведет: "ЦГ"
```
